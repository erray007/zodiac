cmake_minimum_required(VERSION 3.25)
project(zodiac VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
message(FATAL_ERROR "Zodiac supports clang only (found ${CMAKE_CXX_COMPILER_ID}). Use the provided presets.")
endif()
include(CTest)
include(cmake/ZodiacReleaseFlags.cmake)
include(cmake/Dependencies.cmake)
include(cmake/ZodiacWarnings.cmake)
include(cmake/ZodiacModule.cmake)
include(cmake/CaptureReleaseLink.cmake)
include(cmake/ZodiacKitStage.cmake)
if(WIN32)
# The release link line and the toolchain identity for lib/release (the kit's release libraries, staged by zodiac_kit_stage below).
zodiac_capture_release_link("${CMAKE_BINARY_DIR}/lib/release")
endif()
add_library(zodiac_plugin_abi INTERFACE)
target_include_directories(zodiac_plugin_abi INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_subdirectory(src/core)
add_subdirectory(src/release)
add_subdirectory(src/cli)
add_subdirectory(modules)
add_subdirectory(tests)
if(WIN32)
zodiac_kit_stage()
# The rest of the kit layout (spec 8.1): headers for module authors, the manual (an empty folder until docs/manual exists), the example project and the licenses.
install(FILES include/zodiac/plugin.h include/zodiac/plugin.hpp DESTINATION include/zodiac)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/docs/manual")
install(DIRECTORY docs/manual/ DESTINATION docs)
else()
install(CODE "file(MAKE_DIRECTORY \"\$ENV{DESTDIR}\${CMAKE_INSTALL_PREFIX}/docs\")")
endif()
install(DIRECTORY examples/hello DESTINATION examples)
install(FILES LICENSE THIRD_PARTY_LICENSES DESTINATION .)
include(cmake/ZodiacPackage.cmake)
zodiac_package()
endif()
