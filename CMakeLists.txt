cmake_minimum_required(VERSION 3.25)
project(zodiac VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
message(FATAL_ERROR "Zodiac supports clang only (found ${CMAKE_CXX_COMPILER_ID}). Use the provided presets.")
endif()
include(CTest)
include(cmake/Dependencies.cmake)
add_library(zodiac_plugin_abi INTERFACE)
target_include_directories(zodiac_plugin_abi INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_subdirectory(tests)
